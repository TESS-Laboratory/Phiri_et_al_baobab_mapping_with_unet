# -------
# Pansharpening of WorldView mosaics using matching PAN mosaics for our study site
# Requires: OSGeo4W with GDAL 3.10.3
# ------

# Path to the GDAL pansharpening executable.
$gdalPansharpen = "C:\OSGeo4W\apps\Python312\Scripts\gdal_pansharpen.exe"

# Parent folder containing one subfolder per study site.
$rootDir = "C:\Users\cp844\University of Exeter\TESSLab - Chafika Phiri Data\Objective 3 Manuscript\Satellite Data\Data\Mosaicked_Images"

# Each job names its PAN and MS files explicitly, so naming conventions may differ.
# Verify these against the actual filenames on disk before running.
$sites = @(
    @{
        Name = "Kremetart"
        Jobs = @(
            @{
                Label = "feb"
                Pan   = "krem_pan_feb_mosaic.tiff"
                Ms    = "krem_ms_feb_mosaic_coregistered_masked.tif"
            },
            @{
                Label = "july"
                Pan   = "krem_pan_july_mosaic.tiff"
                Ms    = "krem_ms_july_mosaic.tiff"
            }
        )
    },
    @{
        Name = "Mapungubwe"
        Jobs = @(
            @{
                Label = "july"
                Pan   = "maps_pan_july_mosaic.tiff"
                Ms    = "maps_ms_july_mosaic.tiff"
            }
        )
    }
)

# Loop through each study site.
foreach ($site in $sites) {

    # Folder holding this site's mosaics.
    $siteDir = Join-Path $rootDir $site.Name

    # Loop through each pansharpening job for the site.
    foreach ($job in $site.Jobs) {

        # Build full input and output paths.
        $pan    = Join-Path $siteDir $job.Pan
        $ms     = Join-Path $siteDir $job.Ms
        $output = Join-Path $siteDir "$($site.Name)_$($job.Label)_8band_pansharpened.tif"

        Write-Host ""
        Write-Host "Processing $($site.Name) - $($job.Label)"

        # Skip (rather than abort) if the PAN input is missing.
        if (-not (Test-Path $pan)) {
            Write-Warning "PAN not found, skipping: $pan"
            continue
        }

        # Skip if the multispectral input is missing.
        if (-not (Test-Path $ms)) {
            Write-Warning "MS not found, skipping: $ms"
            continue
        }

        # Skip if the output already exists, making re-runs safe.
        if (Test-Path $output) {
            Write-Host "Output already exists, skipping: $output"
            continue
        }

        # Pansharpen the multispectral mosaic using the PAN mosaic.
        # ZSTD provides lossless compression to reduce the output size.
        & $gdalPansharpen `
            -r bilinear `
            -threads ALL_CPUS `
            -co COMPRESS=ZSTD `
            -co ZSTD_LEVEL=9 `
            -co PREDICTOR=2 `
            -co TILED=YES `
            -co BIGTIFF=YES `
            -co NUM_THREADS=ALL_CPUS `
            $pan `
            $ms `
            $output

        # Report failure but continue with remaining jobs.
        if ($LASTEXITCODE -ne 0) {
            Write-Warning "Pansharpening failed: $($site.Name) $($job.Label)"
            continue
        }

        # Report the size of the completed output in gigabytes.
        $sizeGB = (Get-Item $output).Length / 1GB
        Write-Host "$($site.Name) $($job.Label) completed: $([math]::Round($sizeGB, 2)) GB"
    }
}

Write-Host ""
Write-Host "All jobs processed."
